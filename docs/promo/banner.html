private int _temp_gold;
public int gold { get => _temp_gold; set {
  if (_temp_gold == value) return;
  ColorfulLogger.Log($"gold -> {value}", Color.yellow);
  _temp_gold = value; } }
void Awake() { body = GetComponent<Rigidbody>(); }
void OnApplicationQuit() {
  PlayerPrefs.SetInt("Player.gold", gold);
  PlayerPrefs.Save(); }
