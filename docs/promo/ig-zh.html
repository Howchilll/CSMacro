①定义宏
// Macros/LogMacros.cs：宏就是普通 C#
[Macro(MacroSite.Member)]
public static T LOGGED<T>(T initial)
{
    string t = Macro.Type, n = Macro.Name, s = "@_temp_" + n;
    Macro.Emit($"private {t} {s} = {Macro.Arg(0)};");
    Macro.Emit($"public {t} {n} {{ get => {s}; set {{ " +
               $"Log(\"{n}\", value); {s} = value; }} }}");
    return default;
}
②使用 · Player.cs
[@Save] public int gold  = @LOGGED(100);
[@Save] public int level = @LOGGED(1);
void OnApplicationQuit() { @SAVE_ALL(); }
③编译器拿到 · 构建时自动展开
private int _temp_gold = 100;
public int gold { get => _temp_gold;
    set { Log("gold", value); _temp_gold = value; } }
// … level 同理
void OnApplicationQuit() {
    PlayerPrefs.SetInt("Player.gold", gold);
    PlayerPrefs.SetInt("Player.level", level);
    PlayerPrefs.Save(); }
