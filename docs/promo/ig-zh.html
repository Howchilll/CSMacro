①定义宏
// Macros/M.cs：宏就是普通 C#
[Macro(MacroSite.Member)]
public static T GETSET<T>(T target)
{
    var path = Macro.Arg(0);
    if (Macro.ArgSymbol(0) is not { CanWrite: true })
    { Macro.Error($"`{path}` is read-only"); return default; }
    Macro.Emit($"{Macro.Modifiers} {Macro.Type} {Macro.Name} " +
               $"{{ get => {path}; set => {path} = value; }}");
    return default;
}
②使用 · Player.cs
public int Hp   = @GETSET(Data.Player.Hp);
public int Gold = @GETSET(Data.Player.Gold);
public int Max  = @GETSET(Config.MaxHp);  // ✗ 编译报错：只读
③编译器拿到 · 构建时自动展开
