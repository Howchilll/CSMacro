①Define
// Macros/LogMacros.cs: a macro is plain C#
[Macro(MacroSite.Member)]
public static T LOGGED<T>(T initial)
{
    string t = Macro.Type, n = Macro.Name, s = "@_temp_" + n;
    Macro.Emit($"private {t} {s} = {Macro.Arg(0)};");
    Macro.Emit($"public {t} {n} {{ get => {s}; set {{ " +
               $"Log(\"{n}\", value); {s} = value; }} }}");
    return default;
}
②Use · Player.cs
[@Save] public int gold  = @LOGGED(100);
[@Save] public int level = @LOGGED(1);
void OnApplicationQuit() { @SAVE_ALL(); }
③Compiler gets · expanded at build time
